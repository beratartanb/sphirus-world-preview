"""cad223.py <run> --t0= --t1= [--name=] : speed / cadence / step-length agreement in a window (V2.23 'weight from correct stepping').
Uses contact221.py (window-local calibration) for the plants, the evidence rows for positions:
  speed      = capsule path / time in the window (cm/s)
  cadence    = plants of either foot per minute (contacts >= 0.08 s, first contact frame = plant)
  step_gnd   = mean distance between consecutive plants of OPPOSITE feet (ball xy at the plant, along the travel direction)
  step_impl  = speed / cadence (the step the capsule travel implies)
  ratio      = step_gnd / step_impl (1.0 = the feet land where the body travels; < 1 the body out-runs the steps -> skate / stride warp)
  slip       = mean / max net displacement per contact and accumulated drag path (contact221)
  rate       = mean advance of the selected MM time per real second while the same clip stays selected (1.0 = native speed)
Human reference (adult, level): walk 1.2-1.5 m/s @ 100-120 spm, jog 2.5-3.5 m/s @ 150-170 spm, sprint 4.5-5.5 m/s @ 170-190 spm."""
import json, math, sys, pathlib, subprocess
P = next(q for q in pathlib.Path(__file__).resolve().parents if (q / 'sphirus.uproject').exists()); E = P / 'Saved/Codex/SHME_V1_20260925/Evidence'
args = [a for a in sys.argv[1:] if not a.startswith('--')]; opt = dict((a[2:].split('=', 1) + [''])[:2] for a in sys.argv[1:] if a.startswith('--'))
lab = args[0]; T0 = float(opt['t0']); T1 = float(opt['t1'])
subprocess.run([sys.executable, str(P / 'Tools/SHME_V221_20261021/contact221.py'), lab, '--t0=%s' % T0, '--t1=%s' % T1, '--json'], capture_output=True)
C = json.loads((P / 'Saved/Codex/SHME_V216_20261018/contact' / (lab + '.json')).read_text())
R = sorted([r for r in json.load(open(E / (lab + '.json'))) if isinstance(r, dict) and r.get('t') is not None and isinstance(r.get('ax'), list)], key=lambda r: float(r['t']))
W = [r for r in R if T0 <= float(r['t']) <= T1]
def at(t): return min(R, key=lambda r: abs(float(r['t']) - t))
path = sum(math.hypot(b['ax'][0] - a['ax'][0], b['ax'][1] - a['ax'][1]) for a, b in zip(W, W[1:])); dur = float(W[-1]['t']) - float(W[0]['t'])
spd = path / dur if dur > 0 else 0
pl = []
for s in 'lr':
    for c in C['feet'][s]['contacts']:
        if c['dur'] >= 0.08 and T0 + 0.02 < c['t0'] < T1: pl.append((c['t0'], s, at(c['t0'])['vb_ball_' + s], c))
pl.sort(key=lambda x: x[0])
steps = []
for a, b in zip(pl, pl[1:]):
    if a[1] != b[1]:
        tr = (W[-1]['ax'][0] - W[0]['ax'][0], W[-1]['ax'][1] - W[0]['ax'][1]); tn = math.hypot(*tr) or 1
        d = (b[2][0] - a[2][0], b[2][1] - a[2][1]); steps.append(abs(d[0] * tr[0] + d[1] * tr[1]) / tn)
cad = 60.0 * (len(pl) - 1) / (pl[-1][0] - pl[0][0]) if len(pl) > 2 else 0
sg = sum(steps) / len(steps) if steps else 0; si = spd / (cad / 60.0) if cad else 0
nets = [x[3]['net_cm'] for x in pl]; drag = sum(x[3]['drag_path_cm'] for x in pl)
# playback rate of the selected MM clip (same object, monotonic time)
adv = 0.0; real = 0.0
for a, b in zip(W, W[1:]):
    if a.get('av_Phase27_SelectedObject') == b.get('av_Phase27_SelectedObject') and str(b.get('av_CP_StartOwned')) != 'True':
        d_ = float(b.get('av_Phase27_SelectedTime') or 0) - float(a.get('av_Phase27_SelectedTime') or 0); dt_ = float(b['t']) - float(a['t'])
        if 0 <= d_ < 0.2 and dt_ > 0: adv += d_; real += dt_
clips = {}
for r in W: k = str(r.get('av_Phase27_SelectedObject')).rsplit('.', 1)[-1]; clips[k] = clips.get(k, 0) + 1
top = sorted(clips.items(), key=lambda x: -x[1])[:3]
o = {'run': lab, 'name': opt.get('name', ''), 't0': T0, 't1': T1, 'speed': round(spd, 1), 'cadence_spm': round(cad, 1), 'plants': len(pl), 'step_gnd_cm': round(sg, 1), 'step_impl_cm': round(si, 1),
     'ratio': round(sg / si, 3) if si else 0, 'slip_mean_cm': round(sum(nets) / len(nets), 1) if nets else 0, 'slip_max_cm': max(nets) if nets else 0, 'drag_path_cm': round(drag, 1),
     'rate': round(adv / real, 3) if real else 0, 'clips': top}
print('CAD223 %-10s %-34s v %5.1f cm/s  cad %5.1f spm  step gnd %5.1f / impl %5.1f  ratio %.2f  slip %.1f (max %.1f) drag %.1f  rate %.2f  %s' % (o['name'], lab[-34:], o['speed'], o['cadence_spm'], o['step_gnd_cm'], o['step_impl_cm'], o['ratio'], o['slip_mean_cm'], o['slip_max_cm'], o['drag_path_cm'], o['rate'], top[:2]))
print('CAD223_JSON ' + json.dumps(o))
