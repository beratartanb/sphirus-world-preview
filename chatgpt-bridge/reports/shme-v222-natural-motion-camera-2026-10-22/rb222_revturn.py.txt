contact numbers), but the upper body is the vendor sprint-turn performance: trunk pitch up to 32 deg (sprint loop 20-25), roll -15..+20

            difference applied at spine_01; KEEP_TWIST of the source's chest-vs-pelvis yaw is kept (the shoulders still lead the turn).
  arms / neck / head: the loop's locals at the mapped phase (same jog -> sprint blend).
  entry   : source -> new over [ENTRY0, ENTRY1] (smoothstep), so the cut clip's exit pose is still met at the MM entry window (0.52-0.68).
Source keys are cached once in Saved/Codex/SHME_V222_20261022/revturn_src/<clip>.json (every bake starts from the ORIGINAL keys).
Clips are project-owned (SHME/Animations/V216) and baked IN PLACE (file checkpoint in Saved/Codex/SHME_V222_20261022/checkpoint/revturn/).
builtins.RB222 = {'dry_run': True, 'only': None, 'keep_twist': 0.5, 'entry0': 0.50, 'entry1': 0.95, 'v_jog': 250, 'v_sprint': 416}.

def slerp(a, b, t):
    d = sum(a[i] * b[i] for i in range(4))
    if d < 0: b = tuple(-x for x in b); d = -d
    if d > 0.9995: return ae.qnorm(tuple(a[i] + (b[i] - a[i]) * t for i in range(4)))
    th = math.acos(min(d, 1.0)); s = math.sin(th)
    return tuple((math.sin((1 - t) * th) * a[i] + math.sin(t * th) * b[i]) / s for i in range(4))
def lerp(a, b, t): return tuple(a[i] + (b[i] - a[i]) * t for i in range(len(a)))
def sm(x): x = min(1.0, max(0.0, x)); return x * x * (3 - 2 * x)
def yaw_of(q):
    f = ae.qrot(q, (0.0, 1.0, 0.0)); return math.degrees(math.atan2(f[1], f[0]))
class Loop:

        self.clip = ae.Clip(self.seq, KEY + UB); self.L = self.clip.local

    def rel(self, b, f):

        return self.clip.comp(b, f, tmp)[0]
    def legs(self):
        """leg phase table: D = fore-aft separation foot_l - foot_r (root frame, +Y forward), its slope, normalised by the max; mean locals"""
        self.D = [self.rel('foot_l', f)[1] - self.rel('foot_r', f)[1] for f in range(self.n)]
        self.Dmax = max(abs(x) for x in self.D)
        self.DD = [self.D[min(f + 1, self.n - 1)] - self.D[max(f - 1, 0)] for f in range(self.n)]
        self.mean = {}
        for bb in UB:
            q0 = self.L[bb][0][1]; acc = [0.0, 0.0, 0.0, 0.0]; tt = [0.0, 0.0, 0.0]
            for f in range(self.n - 1):
                q = self.L[bb][f][1]; sgn = 1.0 if sum(q[i] * q0[i] for i in range(4)) >= 0 else -1.0
                for i in range(4): acc[i] += sgn * q[i]
                for i in range(3): tt[i] += self.L[bb][f][0][i]
            self.mean[bb] = (tuple(x / (self.n - 1) for x in tt), ae.qnorm(tuple(acc)), self.L[bb][0][2])
    def phase_of(self, dn, slope):
        """loop time whose normalised leg separation / slope sign best match the clip's"""
        best = None
        for f in range(self.n - 1):
            c = (dn - self.D[f] / self.Dmax) ** 2 + (0.35 if (slope > 0) != (self.DD[f] > 0) else 0.0)
            if best is None or c < best[0]: best = (c, f)
        return best[1] / FPS
    def local_at(self, b, tau):
        x = (tau % self.T) * FPS; i = int(math.floor(x)); w = x - i; i = min(i, self.n - 2)
        (t0, q0, s0), (t1, q1, s1) = self.L[b][i], self.L[b][i + 1]
        return (lerp(t0, t1, w), slerp(q0, q1, w), lerp(s0, s1, w))
    def heading_rot(self, b, tau):
        """component rotation of bone b in the loop's root frame at loop time tau"""
        loc = {x: [self.local_at(x, tau)] for x in self.clip.parent if x in self.L}
        q = self.clip.comp(b, 0, loc)[1]; qr = self.clip.comp('root', 0, loc)[1]
        return ae.qnorm(ae.qmul(ae.qinv(qr), q))

def phase_map(events, lp):
    """events [(t_clip, foot)] sorted -> list of (t_clip, tau_loop) anchors, monotonic. A strike of the SAME foot as the previous one
    is a re-plant (pivot / adjusting step), not a new gait step: it is skipped, so the phase moves slowly through a pivot (v1 advanced a
    whole loop period for each -> 0.3 s arm / trunk flail, pitch -10 deg in RevTurn_R_Rfoot)"""
    anchors = []; tau = None; last = None
    for t, s in events:
        if s == last: continue
        last = s
        cand = [h + k * lp.T for k in range(0, 40) for h in lp.hs[s]]
        tau = min(c for c in cand if tau is None or c > tau + 0.05) if tau is not None else min(lp.hs[s])
        anchors.append((t, tau))
    return anchors
def tau_at(anchors, t, lp):
    if len(anchors) < 2: return t
    if t <= anchors[0][0]: (a, ta), (b, tb) = anchors[0], anchors[1]
    elif t >= anchors[-1][0]: (a, ta), (b, tb) = anchors[-2], anchors[-1]
    else:
        i = max(k for k in range(len(anchors)) if anchors[k][0] <= t); (a, ta), (b, tb) = anchors[i], anchors[i + 1]
    return ta + (t - a) * (tb - ta) / max(b - a, 1e-4)
