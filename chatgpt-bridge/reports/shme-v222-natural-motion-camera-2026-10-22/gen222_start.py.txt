    ferr = max(ae.vlen(ae.vsub(clip.comp('foot_' + s, f, local)[0], frames[f][1][s][0])) for f in range(len(frames)) for s in ('l', 'r'))
    drift = 0.0; pen = 0.0
    for s in ('l', 'r'):
        for f in range(1, len(frames)):
            a, b_ = bw(f - 1, s), bw(f, s)
            pen = min(pen, b_[2] - BZ0[s])
            if a[2] < BZ0[s] + 1.0 and b_[2] < BZ0[s] + 1.0: drift += math.hypot(b_[0] - a[0], b_[1] - a[1]) if math.hypot(b_[0] - a[0], b_[1] - a[1]) > 0.05 else 0.0
    pz = [clip.comp('pelvis', f, local)[0][2] - local['root'][f][0][2] for f in range(len(frames))]
    fin = len(frames) - 1
    fe = max(ae.vlen(ae.vsub(ae.vsub(clip.comp('foot_' + s, fin, local)[0], local['root'][fin][0]), IDLE_REL[s][0])) for s in ('l', 'r'))
    he = ae.vlen(ae.vsub(ae.vsub(clip.comp('head', fin, local)[0], local['root'][fin][0]), rel(IDL, 'head')[0]))
    if kind == 'stop':
        e0 = max(ae.vlen(ae.vsub(ae.vsub(clip.comp(bn, 0, local)[0], local['root'][0][0]), rel(loop_local(key * T1 / NPH), bn)[0])) for bn in ('foot_l', 'foot_r', 'head', 'pelvis'))
    else:
        e0 = max(ae.vlen(ae.vsub(ae.vsub(clip.comp(bn, 0, local)[0], local['root'][0][0]), rel(IDL, bn)[0])) for bn in ('foot_l', 'foot_r', 'head', 'pelvis'))
    info.update(foot_err=round(ferr, 2), planted_drift=round(drift, 2), penetration=round(pen, 2), pelvis_min_minus_final=round(min(pz) - pz[-1], 1),
                pelvis_start_minus_final=round(pz[0] - pz[-1], 1), entry_err=round(e0, 2), final_foot_err=round(fe, 2), final_head_err=round(he, 2),
                root_dist=round(ae.vlen(ae.vsub(local['root'][-1][0], local['root'][0][0])), 1))
    return local, info

    for tr in [str(t) for t in AL.get_animation_notify_track_names(seq)]:
        AL.remove_animation_notify_events_by_track(seq, tr)
